# How scripts/doctor.py reads each host tool's version. One line per tool:
# the tool name, then a Python regular expression that the first line of its
# detect_command output (.goneat/tools.yaml) must match in full; the first
# group is the version. A tool in a doctor scope without a line here, or whose
# output does not match, fails the doctor.
git      git version ([0-9]+\.[0-9]+\.[0-9]+)( .*)?
gh       gh version ([0-9]+\.[0-9]+\.[0-9]+) \(.*\)
jq       jq-([0-9]+\.[0-9]+(\.[0-9]+)?)(-[A-Za-z0-9.]+)?
python3  Python ([0-9]+\.[0-9]+\.[0-9]+)
gpg      gpg \(GnuPG\) ([0-9]+\.[0-9]+\.[0-9]+)
minisign minisign ([0-9]+\.[0-9]+(\.[0-9]+)?)
npm      ([0-9]+\.[0-9]+\.[0-9]+)
uv       uv ([0-9]+\.[0-9]+\.[0-9]+)( \(.*\))?
go       go version go([0-9]+\.[0-9]+(\.[0-9]+)?) [a-z0-9]+/[a-z0-9]+
